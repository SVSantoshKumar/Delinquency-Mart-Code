
create or replace TEMPORARY table  DISHA_MART_ANALYTICS.COLLECTION_MART.REALIZED AS (
    SELECT
		TO_DATE('2025-11-30') AS actual_date,
        LTDE.COD_ACCT_NO,
        CASE 
            WHEN LTDE.REPAY_MODE IN (0,21) THEN 'CASH'
            WHEN LTDE.REPAY_MODE = 1 THEN 'CHEQUE'
            WHEN LTDE.REPAY_MODE IN (6,7,8,9,12,13) THEN 'ONLINE'
            ELSE 'NO'
        END AS INSTRUMENT_MODE,
        (LTDE.TOT_INSTLMNT_AMT + LTDE.AMT_SETTLEMENT) AS ALLOCATED_AMOUNT,
        'Realized' AS RCPT_STAT,
        cod_rcpt_no as rcpt_no
    FROM DISHA_L1_HARMONIZED.FLEX_LMS.LN_TXN_DTLS_EXT LTDE

    WHERE LTDE.TOT_INSTLMNT_AMT > 0
      AND LTDE.SETTLEMENT_MODE IN ('A','O','')
      AND (
            CASE 
              WHEN LTDE.REPAY_MODE IN (0,21) THEN 'CASH'
              WHEN LTDE.REPAY_MODE = 1 THEN 'CHEQUE'
              WHEN LTDE.REPAY_MODE IN (6,7,8,9,12,13) THEN 'ONLINE'
              ELSE 'NO'
            END
          ) <> 'NO' and DAT_RCPT = TO_DATE('2025-11-30')
);


create or replace TEMPORARY table  DISHA_MART_ANALYTICS.COLLECTION_MART.UNREALIZED AS (
    SELECT
    TO_DATE('2025-11-30') AS actual_date,
        COD_ACCT_NO,
        CASE WHEN COD_COLL_MODE = '0' THEN 'CASH'
             WHEN COD_COLL_MODE = '1' THEN 'ICHQ'
             ELSE '' END AS INSTRUMENT_MODE,
        AMT_INSTALL_PRE_EMI_INT AS ALLOCATED_AMOUNT,
       CASE WHEN flg_repay_status = 'U' THEN 'Deposited' ELSE ''
           END AS rcpt_stat,
           coll_receipt_no as rcpt_no
    FROM DISHA_L1_HARMONIZED.FLEX_LMS.LN_X_COLLECTION_REGISTER
       where dat_value  = TO_DATE('2025-11-30')
);

create or replace TEMPORARY table  DISHA_MART_ANALYTICS.COLLECTION_MART.FINAL AS (
    SELECT * FROM DISHA_MART_ANALYTICS.COLLECTION_MART.REALIZED
    UNION ALL
    SELECT * FROM DISHA_MART_ANALYTICS.COLLECTION_MART.UNREALIZED
);


create or replace TEMPORARY table  DISHA_MART_ANALYTICS.COLLECTION_MART.FINAL_DELINQ AS (
    SELECT
        F.COD_ACCT_NO,
        F.INSTRUMENT_MODE,
        F.ALLOCATED_AMOUNT,
        F.RCPT_STAT AS RECEIPT_STATUS,
        DM.BRANCH_NAME,
        DM.BRANCH_STATE,
        DM.TEAM,
        DM.COLL_NH,
        DM.COLL_COLL_ZH,
        DM.COLL_AFFORDABLE_ACM,
        DM.TOTAL_DUE,
        cd.LOAN_AMOUNT AS LOAN_AMT,
        DM.TICKET,
        CASE
            WHEN DM.OP_DPD = 0 THEN 'a: 0 - 0'
            WHEN DM.OP_DPD BETWEEN 1 AND 30 THEN 'b: 1 - 30'
            WHEN DM.OP_DPD BETWEEN 31 AND 60 THEN 'c: 31 - 60'
            WHEN DM.OP_DPD BETWEEN 61 AND 90 THEN 'd: 61 - 90'
            WHEN DM.OP_DPD BETWEEN 91 AND 120 THEN 'e: 91 - 120'
            WHEN DM.OP_DPD BETWEEN 121 AND 180 THEN 'f: 121 - 180'
            ELSE 'g: More Than 180'
        END AS ODPD,
        COLLX_BKT,
        CASE WHEN DM.TOTAL_DUE <= F.ALLOCATED_AMOUNT THEN 1 else 0 end as N_FLAG
    FROM DISHA_MART_ANALYTICS.COLLECTION_MART.FINAL F
    LEFT JOIN (select * from DISHA_MART_ANALYTICS.COLLECTION_MART.DELINQUENCY_MART WHERE TO_DATE(start_timestamp) <= TO_DATE('2025-11-30') 
  AND (TO_DATE(end_timestamp) > TO_DATE('2025-11-30') OR TO_DATE(end_timestamp) IS NULL)) DM
  INNER JOIN (
    SELECT * 
    FROM DISHA_L2_CURATED.HISTORICAL_LOAD.COLL_DUELIST 
    WHERE extracted_date = TO_DATE('2025-11-30')
) cd ON TRIM(dm.loan_no) = TRIM(cd.loan_no)
        ON TRIM(F.COD_ACCT_NO) = TRIM(DM.LOAN_NO)
);



SELECT
    TO_DATE('2025-11-30') AS actual_date,
    COD_ACCT_NO,
    BRANCH_NAME,
    BRANCH_STATE,
    TEAM,
    COLL_NH,
    COLL_COLL_ZH,
    COLL_AFFORDABLE_ACM,
    ODPD,
    COLLX_BKT,
    N_FLAG,
    -- CASH
    COUNT(CASE WHEN INSTRUMENT_MODE = 'CASH' THEN 1 END) AS CASH_Count,
    SUM(CASE WHEN INSTRUMENT_MODE = 'CASH' THEN Allocated_Amount ELSE 0 END) AS CASH_Sum,

    -- CHEQUE
    COUNT(CASE WHEN INSTRUMENT_MODE = 'CHEQUE' THEN 1 END) AS CHEQUE_Count,
    SUM(CASE WHEN INSTRUMENT_MODE = 'CHEQUE' THEN Allocated_Amount ELSE 0 END) AS CHEQUE_Sum,

    -- ONLINE
    COUNT(CASE WHEN INSTRUMENT_MODE = 'ONLINE' THEN 1 END) AS ONLINE_Count,
    SUM(CASE WHEN INSTRUMENT_MODE = 'ONLINE' THEN Allocated_Amount ELSE 0 END) AS ONLINE_Sum,

    -- HTS
    COUNT(CASE WHEN TICKET = 'HTS' THEN 1 END) AS HTS_Count,
    SUM(CASE WHEN TICKET = 'HTS' THEN Allocated_Amount ELSE 0 END) AS HTS_Sum,

    -- MTS
    COUNT(CASE WHEN TICKET = 'MTS' THEN 1 END) AS MTS_Count,
    SUM(CASE WHEN TICKET = 'MTS' THEN Allocated_Amount ELSE 0 END) AS MTS_Sum,
    -- SHTS
    COUNT(CASE WHEN TICKET = 'SHTS' THEN 1 END) AS SHTS_Count,
    SUM(CASE WHEN TICKET = 'SHTS' THEN Allocated_Amount ELSE 0 END) AS SHTS_Sum,
-- STS
    COUNT(CASE WHEN TICKET = 'STS' THEN 1 END) AS STS_Count,
    SUM(CASE WHEN TICKET = 'STS' THEN Allocated_Amount ELSE 0 END) AS STS_Sum,
    -- TOTALS
    COUNT(*) AS TOTAL_PAYMENT_COUNT,
    SUM(Allocated_Amount) AS TOTAL_PAYMENT_SUM
FROM DISHA_MART_ANALYTICS.COLLECTION_MART.FINAL_DELINQ
GROUP BY 
   COD_ACCT_NO, BRANCH_NAME, BRANCH_STATE, TEAM,
    COLL_NH, COLL_COLL_ZH, COLL_AFFORDABLE_ACM, COLLX_BKT,N_FLAG,    ODPD
ORDER BY 
   COD_ACCT_NO, BRANCH_NAME, BRANCH_STATE, TEAM,
    COLL_NH, COLL_COLL_ZH, COLL_AFFORDABLE_ACM,
    ODPD;
